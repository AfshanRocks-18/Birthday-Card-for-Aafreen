  <script>
    // Randomly choose which balloon has the surprise
    const winningBalloon = Math.floor(Math.random() * 3);

    function checkBalloon(element, choice) {
      if (choice === winningBalloon) {
        document.getElementById('surprise').classList.remove('hidden');

        confetti({
          particleCount: 150,
          spread: 70,
          origin: { y: 0.6 }
        });

        setTimeout(() => {
          document.getElementById('giftcard').classList.remove('hidden');
        }, 1500);
      } else {
        // teasing messages
        const messages = [
          "Haha, not this one 🎈 Try next!",
          "Close... but no surprise here 😉",
          "Teasing you! Pick another 🎈",
          "Nope, this balloon is empty 🎈"
        ];
        alert(messages[Math.floor(Math.random() * messages.length)]);
        // remove the clicked balloon
        element.style.display = "none";
      }
    }
  </script>
